cmake_minimum_required(VERSION 3.20)
project(ysmlib_physics LANGUAGES C CXX)
option(YSM_PHYSICS_DEBUG_NAMES "Retain WASM function names for maintainer diagnostics" OFF)
set(CMAKE_CXX_STANDARD 17)
set(BULLET_PHYSICS_SOURCE_DIR "${BULLET_SOURCE}")
set(BULLET_VERSION 3.25)
set(INSTALL_LIBS OFF)
set(BUILD_SHARED_LIBS OFF)
add_compile_definitions(BT_NO_PROFILE BT_THREADSAFE=0)
if(CMAKE_SYSTEM_NAME STREQUAL "WASI")
  add_compile_options(-mno-simd128 -fno-exceptions -fno-rtti)
endif()
foreach(component LinearMath BulletCollision BulletDynamics BulletSoftBody)
  add_subdirectory("${BULLET_SOURCE}/src/${component}" "${component}")
endforeach()
if(CMAKE_SYSTEM_NAME STREQUAL "WASI")
  add_executable(physics physics_bridge.cpp)
else()
  add_library(physics STATIC physics_bridge.cpp)
  add_executable(bullet_oracle bullet_oracle.cpp)
  target_include_directories(bullet_oracle PRIVATE "${BULLET_SOURCE}/src")
  target_link_libraries(bullet_oracle PRIVATE BulletSoftBody BulletDynamics BulletCollision LinearMath)
endif()
target_include_directories(physics PRIVATE "${BULLET_SOURCE}/src")
target_link_libraries(physics PRIVATE BulletSoftBody BulletDynamics BulletCollision LinearMath)
if(CMAKE_SYSTEM_NAME STREQUAL "WASI")
  set_target_properties(physics PROPERTIES SUFFIX ".wasm")
  # SDK 34's wasip1 driver rejects -mexec-model; select the same reactor CRT explicitly.
  target_link_libraries(physics PRIVATE "${WASI_SDK_PREFIX}/share/wasi-sysroot/lib/wasm32-wasip1/crt1-reactor.o")
  target_link_options(physics PRIVATE -nostartfiles -Wl,--entry=_initialize -Wl,--export=_initialize
    -Wl,-z,stack-size=1048576 -Wl,--initial-memory=16777216 -Wl,--max-memory=268435456)
  if(YSM_PHYSICS_DEBUG_NAMES)
    target_link_options(physics PRIVATE -Wl,--strip-debug)
  else()
    target_link_options(physics PRIVATE -Wl,--strip-all)
  endif()
  foreach(symbol malloc free ysm_physics_abi ysm_physics_create ysm_physics_destroy ysm_physics_body
      ysm_physics_joint ysm_physics_set_pose ysm_physics_velocity ysm_physics_impulse ysm_physics_step
      ysm_physics_body_count ysm_physics_read_bodies ysm_physics_soft ysm_physics_anchor
      ysm_physics_soft_count ysm_physics_read_soft ysm_physics_read_soft_normals ysm_physics_pin
      ysm_physics_reset_forces ysm_physics_configure_joint ysm_physics_set_kinematic ysm_physics_set_gravity
      ysm_physics_forces ysm_physics_clamp_velocities)
    target_link_options(physics PRIVATE "-Wl,--export=${symbol}")
  endforeach()
endif()
